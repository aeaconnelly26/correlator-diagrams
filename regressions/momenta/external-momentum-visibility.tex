% Count actual emitted draw commands and momentum label nodes in complete
% diagrams. Compare public behavior rather than testing the visibility helper.
\newcount\momentumdrawcount
\newcount\momentumlabelcount
\newcount\baselinedrawcount
\newcount\baselinelabelcount
\newbox\checkdiagram
\makeatletter
\let\originalmomentumlabel\corr@putmomentumlabelwithdistance
\renewcommand{\corr@putmomentumlabelwithdistance}[6]{%
  \global\advance\momentumlabelcount by 1
  \originalmomentumlabel{#1}{#2}{#3}{#4}{#5}{#6}}
\apptocmd{\tikz@installcommands}{%
  \def\draw{\global\advance\momentumdrawcount by 1\path[draw]}%
}{}{\PackageError{external-momentum-visibility}{Cannot instrument TikZ draws}{}}
\makeatother
\newcommand{\measurediagram}[1]{%
  \global\momentumdrawcount=0 \global\momentumlabelcount=0
  \setbox\checkdiagram=\hbox{$#1$}}
\newcommand{\assertcount}[3]{%
  \ifnum#1=#2\relax\else
    \PackageError{external-momentum-visibility}{#3: got \the#1, expected #2}{}%
  \fi}
\newcommand{\assertlabels}[2]{\measurediagram{#2}\assertcount\momentumlabelcount{#1}{Label count}}

\newcommand{\assertsuppressed}[4]{%
  \measurediagram{#1[#2]}%
  \baselinelabelcount=\momentumlabelcount
  \baselinedrawcount=\momentumdrawcount
  \measurediagram{#1[#2,external-momentum-hide={#3}]}%
  \assertcount\momentumlabelcount{\numexpr\baselinelabelcount-#4\relax}{Hidden external label count}%
  \ifnum\momentumdrawcount<\baselinedrawcount\else
    \PackageError{external-momentum-visibility}{Hiding did not remove arrow paths}{}%
  \fi}
