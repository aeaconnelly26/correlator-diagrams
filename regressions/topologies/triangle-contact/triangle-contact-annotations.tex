\newcommand{\tccapturepoint}[1]{%
  \pgfpointanchor{#1}{center}%
  \edef\tcsignature{\tcsignature,\the\pgf@x,\the\pgf@y}}
\renewcommand{\corr@trianglecontactdrawannotation}[3]{%
  \tcoriginaldraw{#1}{#2}{#3}%
  \global\advance\tccount by 1
  \def\tcsignature{}%
  \tccapturepoint{corrTCArrowStart}\tccapturepoint{corrTCArrowEnd}%
  \global\expandafter\let\csname tcarrow\tcphase#1\endcsname\tcsignature
  \tccapturepoint{corrTCLabel}%
  \tccapturepoint{corrTCGapStart}\tccapturepoint{corrTCGapEnd}%
  \edef\tcsignature{\tcsignature,\corr@tclabelanchor,\corr@tclabelxshift,\corr@tclabelyshift,\corr@tcdirection}%
  \global\expandafter\let\csname tc\tcphase#1\endcsname\tcsignature
  \corr@ifstringeq{\tccheckoutside}{yes}{%
    % Independent geometric test: actual emitted arrow points must lie on the
    % opposite side of the supporting edge from the triangle centroid.
    \pgfpointdiff{\pgfpointanchor{#2}{center}}{\pgfpointanchor{#3}{center}}%
    \pgfmathsetmacro{\tcdx}{\pgf@x/1cm}\pgfmathsetmacro{\tcdy}{\pgf@y/1cm}%
    \pgfpointdiff{\pgfpointanchor{#2}{center}}{\pgfpointanchor{corrTCArrowStart}{center}}%
    \pgfmathsetmacro{\tcax}{\pgf@x/1cm}\pgfmathsetmacro{\tcay}{\pgf@y/1cm}%
    \pgfpointdiff{\pgfpointanchor{#2}{center}}{\pgfpointanchor{corrTriCentroid}{center}}%
    \pgfmathsetmacro{\tccx}{\pgf@x/1cm}\pgfmathsetmacro{\tccy}{\pgf@y/1cm}%
    \pgfmathsetmacro{\tcsideproduct}{(\tcdx*\tcay-\tcdy*\tcax)*(\tcdx*\tccy-\tcdy*\tccx)}%
    \ifdim\tcsideproduct pt<0pt\else
      \PackageError{M02}{Slot #1 arrow is not outside the triangle}{}%
    \fi
  }{}%
}
\newcommand{\tcmeasure}[2]{%
  \global\advance\tcdiagramchecks by 1
  \def\tcphase{#1}\global\tccount=0
  \edef\tcmeasureoptions{field={},central={},leg-labels={,,,},
    line=plain,show-momenta=true,momentum-labels={p_1,p_2,p_3,p_4},
    triangle-contact-internal-momenta={k_5,k_6,k_7},#2}%
  \setbox\tcbox=\hbox{\expandafter\TriangleContactCorr\expandafter[\tcmeasureoptions]}}
\newcommand{\tcassertcount}[1]{%
  \ifnum\tccount=#1\relax\else\PackageError{M02}{Expected #1 annotations, got \the\tccount}{}\fi}
\newcommand{\tccompare}[1]{%
  \edef\tctarget{#1}%
  \@for\tccompareslot:={1,2,3,4,5,6,7}\do{%
    \expandafter\let\expandafter\tca\csname tcbaseline\tccompareslot\endcsname
    \expandafter\let\expandafter\tcb\csname tcchanged\tccompareslot\endcsname
    \ifnum\tccompareslot=\tctarget\relax
      \ifx\tca\tcb\PackageError{M02}{Slot \tctarget option had no effect}{}\fi
    \else
      \ifx\tca\tcb\else\PackageError{M02}{Changing slot \tctarget also changed slot \tccompareslot}{}\fi
    \fi}}
\newcommand{\tcassertsame}{%
  \@for\tccompareslot:={1,2,3,4,5,6,7}\do{%
    \expandafter\let\expandafter\tca\csname tcbaseline\tccompareslot\endcsname
    \expandafter\let\expandafter\tcb\csname tcchanged\tccompareslot\endcsname
    \ifx\tca\tcb\else\PackageError{M02}{Unexpected difference in slot \tccompareslot}{}\fi}}
\newcommand{\tcassertarrowunchanged}[1]{%
  \expandafter\let\expandafter\tca\csname tcarrowbaseline#1\endcsname
  \expandafter\let\expandafter\tcb\csname tcarrowchanged#1\endcsname
  \ifx\tca\tcb\else\PackageError{M02}{Label-only control moved slot #1 arrow}{}\fi}
\newcommand{\tcassertvalue}[2]{%
  \pgfmathsetmacro{\tcdelta}{abs((#1)-(#2))}%
  \ifdim\tcdelta pt<0.0001pt\else\PackageError{M02}{Explicit grouped setting ignored: #1 vs #2}{}\fi}
\makeatother
