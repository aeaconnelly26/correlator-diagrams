\newcommand{\framecheck}[1]{%
 \coordinate (frameExpectedUL) at (-\framex,\framey);
 \coordinate (frameExpectedLL) at (-\framex,-\framey);
 \coordinate (frameExpectedUR) at (\framex,\framey);
 \coordinate (frameExpectedLR) at (\framex,-\framey);
 \rtclose{#1UL}{frameExpectedUL}\rtclose{#1LL}{frameExpectedLL}%
 \rtclose{#1UR}{frameExpectedUR}\rtclose{#1LR}{frameExpectedLR}%
}

\newcommand{\framemeasure}[1]{%
 \global\advance\framecases by1

 \@for\frametopology:={t-tree,s-tree,u-tree,s,t,u,contact,box,cross-box}\do{\framemeasure{topology=\frametopology}}%
 \@for\framepair:={horizontal,vertical,crossed}\do{\framemeasure{topology=contact,contact-pairing=\framepair}}%
 \@for\frameside:={top,bottom,left,right}\do{\framemeasure{topology=triangle-contact,triangle-contact-orientation=\frameside}}%
 \@for\framechannel:={t-tree,s-tree}\do{%
  \@for\frameinsertion:={bubble,seagull,swordfish}\do{%
   \@for\frameslot:={1,2,3,4}\do{\framemeasure{topology=\framechannel,insertion=\frameinsertion,insertion-slot=\frameslot}}}}
 \framemeasure{topology=t-tree,insertion=triangle,tree-triangle-side=top}%
 \framemeasure{topology=t-tree,insertion=triangle,tree-triangle-side=bottom}%
 \framemeasure{topology=s-tree,insertion=triangle,tree-triangle-side=left}%
 \framemeasure{topology=s-tree,insertion=triangle,tree-triangle-side=right}%
}
