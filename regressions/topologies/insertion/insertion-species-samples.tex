\newbox\speciesprobe

\makeatletter
\let\speciesoldbubblecoords\corr@setttreebubbleinsertioncoords
\renewcommand{\corr@setttreebubbleinsertioncoords}{%
  \speciesoldbubblecoords
  \pgfpointdiff{\pgfpointanchor{corrInsertionA}{center}}{\pgfpointanchor{corrInsertionB}{center}}%
  \pgfmathsetlengthmacro{\specieslength}{veclen(\pgf@x,\pgf@y)}\xdef\speciesspan{\specieslength}%
  \pgfpointdiff{\pgfpointanchor{corrInsertionOuter}{center}}{\pgfpointanchor{corrInsertionA}{center}}%
  \pgfmathsetlengthmacro{\specieslength}{veclen(\pgf@x,\pgf@y)}\xdef\speciespre{\specieslength}%
  \pgfpointdiff{\pgfpointanchor{corrInsertionB}{center}}{\pgfpointanchor{corrInsertionHost}{center}}%
  \pgfmathsetlengthmacro{\specieslength}{veclen(\pgf@x,\pgf@y)}\xdef\speciespost{\specieslength}%
}
\newcommand{\speciesmeasure}[4]{%
  \global\advance\speciescases by 1
  \setbox\speciesprobe=\hbox{#1[field={},central={},leg-labels={,,,},line=plain,
    show-momenta=false,insertion=#2,insertion-slot=#3,#4]}}
\newcommand{\speciesnear}[3]{%
  \pgfmathsetlengthmacro{\specieserror}{abs(#1-(#2))}%
  \ifdim\specieserror>0.03pt\PackageError{Insertion size}{#3}{}\fi}
\makeatother

\makeatletter
\@for\specieshost:={T,S}\do{%
  \expandafter\let\expandafter\speciesmacro\csname\specieshost TreeChannelCorr\endcsname
  \@for\speciesslot:={1,2,3,4}\do{%
    \speciesmeasure\speciesmacro{bubble}{\speciesslot}{}%
    \xdef\speciescompactspan{\speciesspan}\xdef\speciescompactpre{\speciespre}\xdef\speciescompactpost{\speciespost}%
    \speciesmeasure\speciesmacro{bubble}{\speciesslot}{insertion-scale=0.66}%
    \speciesnear{\speciescompactspan}{\speciesspan*0.50/0.66}{Default bubble did not shrink proportionally}%
    % The released .66 footprint is recoverable and both host pieces gain room.
    \speciesnear{\speciesspan}{2*0.74*0.66*1cm}{Explicit former bubble size was overridden}%
    \pgfmathsetlengthmacro{\speciesgain}{(\speciesspan-\speciescompactspan)/2}%
    \speciesnear{\speciescompactpre-\speciespre}{\speciesgain}{Outer host segment did not gain the released space}%
    \speciesnear{\speciescompactpost-\speciespost}{\speciesgain}{Inner host segment did not gain the released space}%
    \speciesmeasure\speciesmacro{bubble}{\speciesslot}{}%
    \speciesnear{\speciesspan}{\speciescompactspan}{Former-size override leaked into next bubble}%
    \speciesmeasure\speciesmacro{bubble}{\speciesslot}{insertion-scale=0.42}%
    \speciesnear{\speciesspan}{\speciescompactspan*0.42/0.50}{Custom bubble size was overridden}%
    \speciesmeasure\speciesmacro{seagull}{\speciesslot}{}%
    \speciesnear{\corr@seagullloopradius}{0.36*1.55*0.66*1cm}{Bubble override changed the default seagull size}%
    \speciesmeasure\speciesmacro{seagull}{\speciesslot}{insertion-scale=0.42}%
    \speciesnear{\corr@seagullloopradius}{0.36*1.55*0.42*1cm}{Explicit seagull size was overridden}%
    \speciesmeasure\speciesmacro{bubble}{\speciesslot}{}%
    \speciesnear{\speciesspan}{\speciescompactspan}{Seagull override leaked into next bubble}%
  }}%
\typeout{PASS insertion size: \the\speciescases\space public diagrams; shorter bubble span, longer host segments, explicit sizes and state reset}%
\let\corr@setttreebubbleinsertioncoords\speciesoldbubblecoords
% Verify emitted central labels disappear independently of external/loop labels,
% and the next public diagram returns to the compatible visible default.
\newcount\speciesvisibilitycases \newcount\specieslabels
\let\speciesoldlabel\corr@putmomentumlabelwithdistance
\renewcommand{\corr@putmomentumlabelwithdistance}[6]{%
  \global\advance\specieslabels by 1
  \speciesoldlabel{#1}{#2}{#3}{#4}{#5}{#6}}
\newcommand{\speciesvisibility}[3]{%
  \global\advance\speciesvisibilitycases by 1\global\specieslabels=0
  \edef\speciesalloptions{field={},central={},leg-labels={,,,},line=plain,
    show-momenta=true,momentum-labels={p_1,p_2,p_3,p_4,P},#2}%
  \setbox\speciesprobe=\hbox{\expandafter#1\expandafter[\speciesalloptions]}%
  \ifnum\specieslabels=#3\relax\else
    \PackageError{Central momentum visibility}{Emitted label count \the\specieslabels, expected #3}{}\fi}
\@for\speciesmacro:={\STreeChannelCorr,\TTreeChannelCorr}\do{%
  \@for\specieskind:={bubble,seagull}\do{%
    \@for\speciesslot:={1,2,3,4}\do{%
      \edef\speciesoptions{insertion=\specieskind,insertion-slot=\speciesslot,
        bubble-momenta={,k},seagull-loop-momentum=k}%
      \speciesvisibility\speciesmacro{\speciesoptions,show-channel-momentum=false}{6}%
      \speciesvisibility\speciesmacro{\speciesoptions}{7}%
    }}}
\@for\speciesmacro:={\STreeChannelCorr,\TTreeChannelCorr,\UTreeChannelCorr}\do{%
  \speciesvisibility\speciesmacro{show-channel-momentum=false}{4}%
  \speciesvisibility\speciesmacro{}{5}%
  \speciesvisibility\speciesmacro{show-channel-momentum=false,show-channel-momentum=true}{5}%
}
\let\corr@putmomentumlabelwithdistance\speciesoldlabel
\typeout{PASS central momentum visibility: \the\speciesvisibilitycases\space public diagrams; external/loop labels retained; visible defaults and explicit override reset}
\makeatother
