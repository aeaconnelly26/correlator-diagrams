\newcommand{\rvmeasure}[2]{%
  \global\advance\rvcases by 1
  \@for\rvidx:={1,2,3,4}\do{\expandafter\gdef\csname rvdir\rvidx\endcsname{missing}}%
  \setbox\rvbox=\hbox{#1[field={},central={},leg-labels={,,,},line=plain,channel-line=plain,
    show-momenta=true,momentum-labels={p_1,p_2,p_3,p_4,P},#2]}%
  \@for\rvidx:={1,2,3,4}\do{%
    \ifnum\rvidx<3\let\rvwanted\rvexpectedleft\else\let\rvwanted\rvexpectedright\fi
    \edef\rvactual{\csname rvdir\rvidx\endcsname}%
    \ifx\rvactual\rvwanted\else\PackageError{M04-review}{Slot \rvidx: got \rvactual; expected \rvwanted}{}\fi}%
}
