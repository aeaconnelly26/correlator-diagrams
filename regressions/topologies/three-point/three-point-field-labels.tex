\newcommand{\FFCheckDistance}[1]{%
  \ifdim#1<-.02pt\PackageError{field-label-check}{Unexpected field-label displacement}{}\fi
  \ifdim#1>.02pt\PackageError{field-label-check}{Unexpected field-label displacement}{}\fi
  \global\advance\FFChecks by 1
}
\newcommand{\FFCaptureFields}{%
  \@for\FFSlot:={One,Two,Three}\do{%
    \pgfpointtransformed{\pgfpointanchor{FFField\FFSlot}{center}}%
    \ifFFRecord
      \global\csname FFBaseX\FFSlot\endcsname=\pgf@x
      \global\csname FFBaseY\FFSlot\endcsname=\pgf@y
    \else
      \@tempdima=\dimexpr\pgf@x-\csname FFBaseX\FFSlot\endcsname-\csname FFDX\FFSlot\endcsname\relax
      \FFCheckDistance{\@tempdima}%
      \@tempdima=\dimexpr\pgf@y-\csname FFBaseY\FFSlot\endcsname-\csname FFDY\FFSlot\endcsname\relax
      \FFCheckDistance{\@tempdima}%
    \fi
  }%
}
