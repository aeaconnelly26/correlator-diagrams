\newcommand{\MockCurvedMomentum}[9]{%
  \pgfmathsetmacro{\mockmomstep}{(#5-#4)/8}%
  \pgfmathsetmacro{\mockmompone}{#4+\mockmomstep}%
  \pgfmathsetmacro{\mockmomptwo}{#4+2*\mockmomstep}%
  \pgfmathsetmacro{\mockmompthree}{#4+3*\mockmomstep}%
  \pgfmathsetmacro{\mockmompfour}{#4+4*\mockmomstep}%
  \pgfmathsetmacro{\mockmompfive}{#4+5*\mockmomstep}%
  \pgfmathsetmacro{\mockmompsix}{#4+6*\mockmomstep}%
  \pgfmathsetmacro{\mockmompseven}{#4+7*\mockmomstep}%
  \path
    #1 to[#3]
    coordinate[pos=#4] (mockmomstart)
    coordinate[pos=\mockmompone] (mockmompone)
    coordinate[pos=\mockmomptwo] (mockmomptwo)
    coordinate[pos=\mockmompthree] (mockmompthree)
    coordinate[pos=\mockmompfour] (mockmompfour)
    coordinate[pos=\mockmompfive] (mockmompfive)
    coordinate[pos=\mockmompsix] (mockmompsix)
    coordinate[pos=\mockmompseven] (mockmompseven)
    coordinate[pos=#6] (mockmomlabel)
    coordinate[pos=#5] (mockmomend)
    #2;
  \draw[-{Stealth[length=3.8pt]},line width=\corr@momentumlinewidth,line cap=round,line join=round]
    (mockmomstart) --
    (mockmompone) --
    (mockmomptwo) --
    (mockmompthree) --
    (mockmompfour) --
    (mockmompfive) --
    (mockmompsix) --
    (mockmompseven) --
    (mockmomend);
  \corr@putmomentumlabelwithdistance
    {(mockmomlabel)}
    {#7}
    {#8}
    {0pt}
    {0pt}
    {#9}%
}
\newcommand{\MockStraightMomentum}[5]{%
  \path #1 -- coordinate[pos=#3] (mockstraightlabel) #2;
  \draw[-{Stealth[length=3.6pt]},line width=\corr@momentumlinewidth,line cap=round]
    #1 -- #2;
  \corr@putmomentumlabelwithdistance
    {(mockstraightlabel)}
    {above}
    {#4}
    {0pt}
    {0pt}
    {#5}%
}

}
