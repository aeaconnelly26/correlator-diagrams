\newcommand{\diagramcell}[2]{%
  \begin{minipage}[t]{0.31\linewidth}
  \centering
  \textbf{#1}\par\vspace{0.8em}
  #2
  \end{minipage}%
}

\newcommand{\markerline}[3]{%
  \begin{tikzpicture}
    \begin{feynhand}
      \vertex (a) at (0,0);
      \vertex (b) at (#2,0);
      \propag[#1] (a) to (b);
    \end{feynhand}
    \filldraw[draw=black,fill=black] (a) circle[radius=2.15pt];
    \filldraw[draw=black,fill=black] (b) circle[radius=2.15pt];
  \end{tikzpicture}\par
  {\small #3}
}

\newcommand{\diagonalendpointline}[2]{%
  \begin{tikzpicture}
    \begin{feynhand}
      \vertex (a) at (0,0);
      \vertex (b) at (1.75,0.9);
      \propag[#1] (a) to (b);
    \end{feynhand}
    \filldraw[draw=black,fill=black] (a) circle[radius=2.15pt];
    \filldraw[draw=black,fill=black] (b) circle[radius=2.15pt];
  \end{tikzpicture}\par
  {\small #2}
}
