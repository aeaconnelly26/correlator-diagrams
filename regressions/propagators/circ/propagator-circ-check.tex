\newcommand{\linecheck}[2]{%
  \begin{tikzpicture}
    \begin{feynhand}
      \vertex (a) at (0,0);
      \vertex (b) at (2.25,0);
      \propag[#1] (a) to (b);
    \end{feynhand}
  \end{tikzpicture}\par
  {\small #2}
}

\newcommand{\procaendpointcheck}[2]{%
  \begin{tikzpicture}
    \begin{feynhand}
      \vertex (a) at (0,0);
      \vertex (b) at (2.25,0);
      \propag[#1] (a) to (b);
    \end{feynhand}
    \filldraw[draw=black,fill=black] (a) circle[radius=1.05pt];
    \filldraw[draw=black,fill=black] (b) circle[radius=1.05pt];
  \end{tikzpicture}\par
  {\small #2}
}
