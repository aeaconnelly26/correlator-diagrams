\newcommand{\FFCaptureLabel}[2]{%
  \pgfpointtransformed{\pgfpointanchor{FFObservedLabel}{center}}%
  \ifFFRecord
    \global\csname FFBaseX#2\endcsname=\pgf@x
    \global\csname FFBaseY#2\endcsname=\pgf@y
  \else
    \@tempdima=\dimexpr\pgf@x-\csname FFBaseX#2\endcsname-\csname FFDX#2\endcsname\relax
    \FFCheckDistance{\@tempdima}%
    \@tempdima=\dimexpr\pgf@y-\csname FFBaseY#2\endcsname-\csname FFDY#2\endcsname\relax
    \FFCheckDistance{\@tempdima}%
  \fi
  \pgfpointtransformed{\pgfpointanchor{#1}{center}}%
  \ifFFRecord
    \global\csname FFAnchorX#2\endcsname=\pgf@x
    \global\csname FFAnchorY#2\endcsname=\pgf@y
  \else
    \@tempdima=\dimexpr\pgf@x-\csname FFAnchorX#2\endcsname\relax
    \FFCheckDistance{\@tempdima}%
    \@tempdima=\dimexpr\pgf@y-\csname FFAnchorY#2\endcsname\relax
    \FFCheckDistance{\@tempdima}%
  \fi
}
